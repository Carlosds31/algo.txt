algorithme: analyse_phrase
variables: car: caractere; 
           log, mot, voyelle: entier    
DEBUT
log:= 0
mot:= 1
voyelle:= 0
ecrire ("Saisir une phrase")
lire ("car") 
// détermination de la longueur
Tant que (car != ".") Faire
    log:= log + 1
// détermine le nombre de mot 
Si (car=" ") Alors
    mot:= mot + 1
    Finsi
// Déterminé le nombre de voyelle
Si (car= "a" ou car= "e" ou car= "i" ou car= "o" ou car= "u" ou car= "A" ou car= "E" ou car= "I" ou car= "O" ou car= "U") Alors
     voyelle:= voyelle + 1
    Finsi
    lire("car")
    ecrire("La longueur de la phrase est :",log)
    ecrire("Le nombre de mot dans la phrase est :",mot)
    ecrire("Le nombre de voyelle dans la phrase est :",voyelle)
FinFaire
